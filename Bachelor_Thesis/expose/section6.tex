\section{Thesis Timeline}

\begin{enumerate}
    \item \textbf{Week 1:} Literature review \& theoretical background, finalize methods (traditional ML \& lightweight CNN)
    \item \textbf{Week 2:} Data preparation, image segmentation \& preprocessing, data augmentation experiments
    \item \textbf{Week 3:} Implement traditional ML methods, training and preliminary testing
    \item \textbf{Week 4:} Implement lightweight CNN, low-sample training, preliminary performance comparison
    \item \textbf{Week 5:} Full experiments (2-class / 3-class), record metrics (Accuracy, Precision, Recall, F1-score, Confusion Matrix)
    \item \textbf{Week 6:} Analyze experiment results, create figures and tables, write methodology and experiments sections
    \item \textbf{Week 7:} Complete thesis first draft (Introduction, Theoretical Background, Methodology, Experiments \& Results)
    \item \textbf{Week 8:} Thesis revision \& polishing, check formatting and references, prepare for submission
\end{enumerate}

\noindent
\textit{Note: If a given week's tasks require more time, the week may be extended by one week, with subsequent tasks adjusted accordingly.}
